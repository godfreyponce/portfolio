- [ ] **Step 1: Write the generator**

`$SCRATCH/gen_svg.py`:

```python
#!/usr/bin/env python3
"""Render braille portrait + neofetch panel to a GitHub-themed SVG.
usage: gen_svg.py <classic|stacked|terminal> <dark|light> <out.svg>"""
import html, os, sys

SCRATCH = os.environ["SCRATCH"]
ART = open(f"{SCRATCH}/portrait_d.txt").read().rstrip("\n").split("\n")   # 51 x 96
PANEL = open(f"{SCRATCH}/panel.txt").read().rstrip("\n").split("\n")

FONT = "Consolas, Menlo, monospace"
ADV = 8.8          # px per portrait cell @ 16px — TUNE in the render check (Task 2 Step 3)
P_SIZE, P_LH = 16, 16          # portrait: font-size, line-height 1 (tight, so cells tile)
T_SIZE, T_LH = 16, 22          # panel text
ART_W, ART_H = 96 * ADV, 51 * P_LH

THEMES = {
  "dark":  dict(canvas="#0d1117", card="#161b22", border="#30363d",
                ink="#c9d1d9", key="#ffa657", val="#a5d6ff"),
  "light": dict(canvas="#ffffff", card="#f6f8fa", border="#d0d7de",
                ink="#24292f", key="#953800", val="#0a3069"),
}

def esc(s): return html.escape(s, quote=False)

def text_block(lines, x, y, size, lh, cls_lines=False):
    out = [f'<text xml:space="preserve" x="{x}" y="{y}" font-size="{size}">']
    for i, ln in enumerate(lines):
        dy = lh if i else 0
        if cls_lines and ":" in ln:
            k, v = ln.split(":", 1)
            out.append(f'<tspan x="{x}" dy="{dy}"><tspan class="k">{esc(k)}:</tspan>'
                       f'<tspan class="v">{esc(v)}</tspan></tspan>')
        else:
            out.append(f'<tspan x="{x}" dy="{dy}" class="i">{esc(ln)}</tspan>')
    out.append("</text>")
    return "\n".join(out)

def build(layout, theme):
    t = THEMES[theme]
    pad = 24
    body = []
    if layout == "classic":       # portrait left, panel right — the Andrew6rant shape
        W = pad + ART_W + 56 + 52 * ADV + pad
        H = pad + ART_H + pad
        panel_y = (H - len(PANEL) * T_LH) / 2 + T_SIZE
        body.append(f'<rect width="{W}" height="{H}" rx="8" fill="{t["card"]}" stroke="{t["border"]}"/>')
        body.append(text_block(ART, pad, pad + P_SIZE, P_SIZE, P_LH))
        body.append(text_block(PANEL, pad + ART_W + 56, panel_y, T_SIZE, T_LH, cls_lines=True))
    elif layout == "stacked":     # portrait on top, panel centered below — minimalist
        W = pad + ART_W + pad
        H = pad + ART_H + 32 + len(PANEL) * T_LH + pad
        panel_x = (W - 50 * ADV) / 2
        body.append(f'<rect width="{W}" height="{H}" fill="{t["canvas"]}"/>')
        body.append(text_block(ART, pad, pad + P_SIZE, P_SIZE, P_LH))
        body.append(f'<line x1="{pad}" y1="{pad+ART_H+16}" x2="{W-pad}" y2="{pad+ART_H+16}" stroke="{t["border"]}"/>')
        body.append(text_block(PANEL, panel_x, pad + ART_H + 32 + T_SIZE, T_SIZE, T_LH, cls_lines=True))
    elif layout == "terminal":    # classic geometry inside macOS-terminal chrome + prompt line
        bar = 36
        W = pad + ART_W + 56 + 52 * ADV + pad
        H = bar + pad + T_LH + ART_H + pad
        body.append(f'<rect width="{W}" height="{H}" rx="10" fill="{t["card"]}" stroke="{t["border"]}"/>')
        body.append(f'<rect width="{W}" height="{bar}" rx="10" fill="{t["canvas"]}"/>')
        for i, c in enumerate(("#ff5f57", "#febc2e", "#28c840")):
            body.append(f'<circle cx="{20+i*22}" cy="{bar/2}" r="6" fill="{c}"/>')
        body.append(text_block([ "geov@github:~$ neofetch" ], pad, bar + pad + T_SIZE, T_SIZE, T_LH, cls_lines=False))
        body.append(text_block(ART, pad, bar + pad + T_LH + P_SIZE, P_SIZE, P_LH))
        panel_y = bar + pad + T_LH + (ART_H - len(PANEL) * T_LH) / 2 + T_SIZE
        body.append(text_block(PANEL, pad + ART_W + 56, panel_y, T_SIZE, T_LH, cls_lines=True))
    else:
        sys.exit(f"unknown layout {layout}")
    style = (f'text{{font-family:{FONT};white-space:pre}}'
             f'.i{{fill:{t["ink"]}}} .k{{fill:{t["key"]};font-weight:600}} .v{{fill:{t["val"]}}}')
    return (f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {W:.0f} {H:.0f}" '
            f'width="{W:.0f}" height="{H:.0f}">\n<style>{style}</style>\n'
            + "\n".join(body) + "\n</svg>\n")

if __name__ == "__main__":
    layout, theme, out = sys.argv[1:4]
    open(out, "w").write(build(layout, theme))
    print(out)
```
